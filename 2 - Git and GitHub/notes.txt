# Configuring Git
git config --global user.name "username"
git config --global user.email "email"
git config user.email
git config user.name

#Legends
U - Untraked
A - Added
M - Modified

# Check the status of all the files and folders
git status

#Git 3 Stage Architecture
1. Working Area
2. Staging Area
3. Commit Stage

# Add all the modified files in the staging area
git add .

# Create a commit
git commit -m "This is the description."

# Show the commit logs
git log

#Show shortened commit log
git log --oneline

Branching

#List all the branches available
git branch -a

# Create a new branch
git branch "branch_name"

#Switch to a different branching
git switch "branch_name"

# Merging branches
git merge "branch_name"

# Delete a branch
git branch -d "branch_name"

# Connect the local repository to the remote repository
git remote add origin [repository-url]

# Rename the master branch to main
git branch -M main

# Initially create a push
git push -u origin main

# Push the code from local to remote
git push
